def call(boolean checkOnTimeout = false) {

    /* ---------------------------------------------------------
       Load environment configuration
    ---------------------------------------------------------- */
    def envProperty = loadEnvironmentProperties()

    def fortifyApiUrl             = envProperty.fortify_api_url
    def fortifyApiToken           = envProperty.fortify_api_credential_id
    def fortifyScanCountThreshold = envProperty.fortify_scan_count_threshold as Integer
    def alertEnabled              = envProperty.fortify_alert_enabled as Boolean

    def devopsWorkflowUrl         = envProperty.devops_workflow_url
    def fortifyWorkflowUrl        = envProperty.fortify_workflow_url
    def devopsChannelEmails       = envProperty.to_email_address_list
    def fortifyChannelEmails      = envProperty.fortify_team_emails

    /* ---------------------------------------------------------
       IMPORTANT: Use a GLOBAL state file (not workspace)
    ---------------------------------------------------------- */
    def alertStateFile = "/var/jenkins_home/fortify-alert-state/${env.JOB_NAME}.state"

    if (!alertEnabled) {
        println "INFO: Fortify alerts are disabled. Skipping alert evaluation."
        return
    }

    /* ---------------------------------------------------------
       Get pending Fortify scan count
    ---------------------------------------------------------- */
    def fortifyScanCount = getFortifyPendingJobsCount(
        fortifyApiToken,
        fortifyApiUrl
    )

    println "INFO: Current Fortify pending scan count: ${fortifyScanCount}"

    /* ---------------------------------------------------------
       LOCKED SECTION – ONLY ONE BUILD AT A TIME
    ---------------------------------------------------------- */
    lock(resource: "fortify-alert-lock") {

        /* ------------------ TIMEOUT PATH ------------------- */
        if (checkOnTimeout) {

            if (fortifyScanCount > fortifyScanCountThreshold) {

                boolean alertAlreadySent = false

                if (fileExists(alertStateFile)) {
                    def state = readFile(alertStateFile).trim()
                    alertAlreadySent = state.equalsIgnoreCase("true")
                }

                if (!alertAlreadySent) {

                    println "WARN: Fortify scan timed out and pending count " +
                            "(${fortifyScanCount}) exceeded threshold " +
                            "(${fortifyScanCountThreshold}). Sending alert."

                    sendTeamsNotification(
                        devopsWorkflowUrl,
                        devopsChannelEmails,
                        "DevOps"
                    )

                    sendTeamsNotification(
                        fortifyWorkflowUrl,
                        fortifyChannelEmails,
                        "Fortify"
                    )

                    writeFile(
                        file: alertStateFile,
                        text: "true\n"
                    )

                } else {
                    println "INFO: Alert already sent. Suppressing duplicate."
                }

            } else {
                println "INFO: Timeout occurred but pending count within threshold. No alert sent."
            }

            return
        }

        /* ---------------- NON-TIMEOUT PATH ----------------- */
        if (fortifyScanCount <= fortifyScanCountThreshold) {

            println "INFO: Pending scans back within threshold. Resetting alert state."

            writeFile(
                file: alertStateFile,
                text: "false\n"
            )

        } else {
            println "INFO: Pending scans exceeded threshold but no timeout. No alert sent."
        }
    }
}

/* =========================================================
   Fortify API – Pending scan count
========================================================= */
def getFortifyPendingJobsCount(def fortifyApiToken, def fortifyApiUrl) {

    withCredentials([
        string(credentialsId: fortifyApiToken, variable: 'FORTIFY_TOKEN')
    ]) {

        def response = sh(
            script: """
                curl -ks \\
                -H "Authorization: FortifyToken ${FORTIFY_TOKEN}" \\
                "${fortifyApiUrl}/cloudjobs?fields=jobState&q=jobState:PENDING"
            """,
            returnStdout: true
        ).trim()

        def json = readJSON(text: response)
        def count = json?.count ?: 0

        println "INFO: Pending scans reported by Fortify API: ${count}"
        return count
    }
}
