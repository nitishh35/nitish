def call() {
    try {
        echo "INFO: Preparing Teams notification payload"

        // -------------------------------
        // BUILD STATUS + FAILED STAGE
        // -------------------------------
        def buildStatus = currentBuild.result ?: "SUCCESS"
        def failedStage = (buildStatus == "FAILURE") ? env.failedStage : "NA"

        // -------------------------------
        // USER DETAILS
        // -------------------------------
        def userInfo = getBuildTriggeredUserDetails()
        def userName = userInfo.userName ?: "Unknown User"
        def userEmail = userInfo.userEmail ?: "Unknown Email"
        def buildUrl = env.BUILD_URL ?: "Unknown URL"

        if (userName == 'SVC-APP-RLCT') {
            echo "INFO: Notification suppressed for automation user: ${userName}"
            return
        }

        // -------------------------------
        // FINAL PAYLOAD
        // -------------------------------
        def buildData = [
            pipelineURL      : buildUrl,
            triggeredBy      : userName,
            triggeredByEmail : userEmail,
            status           : buildStatus,
            stage            : failedStage
        ]

        def bodyJson = groovy.json.JsonOutput.toJson(buildData)

        echo "INFO: JSON Payload Prepared:"
        echo bodyJson

        // -------------------------------
        // CHANNEL URL LOOKUP
        // -------------------------------
        def channelUrl = getProductWorkflowChannelUrl()
        if (!channelUrl) {
            echo "WARN: No Teams channel URL found. Skipping notification."
            return
        }

        // -------------------------------
        // SEND TO TEAMS
        // -------------------------------
        notifyTeam(bodyJson, channelUrl)

    } catch (Exception e) {
        echo "ERROR: Failed to send Teams notification: ${e.message}"
    }
}


// ----------------------------------------------------------
// HTTP POST TO TEAMS
// ----------------------------------------------------------
def notifyTeam(String json, String channelUrl) {
    def response = sh(
        script: """
            curl -k -s -w '%{http_code}' \\
                -H 'Content-Type: application/json' \\
                -X POST '${channelUrl}' \\
                -d '${json.replace("'", "\\'")}'
        """,
        returnStdout: true
    ).trim()

    echo "INFO: Teams Response Code: ${response}"
}
