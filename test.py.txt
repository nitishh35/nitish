node {

    stage('Checkout') {
        checkout scm
    }

    stage('Validate Java Version') {

        def jdkHome = tool name: 'jdk-21', type: 'hudson.model.JDK'

        withEnv([
            "JAVA_HOME=${jdkHome}",
            "PATH=${jdkHome}/bin:${env.PATH}"
        ]) {
            sh '''
                echo "==== JAVA ENV DETAILS ===="
                echo "JAVA_HOME=$JAVA_HOME"
                which java

                echo "==== JAVA VERSION ===="
                java -version
            '''
        }
    }

    stage('Result') {
        echo 'JDK 21 validation completed successfully'
    }
}

def call(def pcfOrg, def pcfSpace) {

    // DEFAULT: assume all foundations are available
    def dev2dal   = "YES"
    def dev2phx   = "YES"
    def dev1east1 = "YES"
    def dev1east2 = "YES"

    // Fetch foundation availability JSON
    def foundationData = getPcfFoundationAvailability()

    // If real data returned, convert Y/N to YES/NO
    if (foundationData && !foundationData.toString().equalsIgnoreCase("NA")) {
        dev2dal   = foundationData.DEV2DAL?.equalsIgnoreCase('N')   ? "NO" : "YES"
        dev2phx   = foundationData.DEV2PHX?.equalsIgnoreCase('N')   ? "NO" : "YES"
        dev1east1 = foundationData.DEV1EAST1?.equalsIgnoreCase('N') ? "NO" : "YES"
        dev1east2 = foundationData.DEV1EAST2?.equalsIgnoreCase('N') ? "NO" : "YES"
    }

    // Track if anything got deployed
    def deployed = false

    // -------------------------------
    // Deploy to ALL available foundations
    // -------------------------------
    if (dev2dal == "YES") {
        println "INFO: Deploying to DEV2-DAL foundation"
        staticFileDeployPCFWithManifest(pcfOrg, pcfSpace, 'api.sys.dev2.dal.pcf.syfbank.com')
        deployed = true
    }

    if (dev2phx == "YES") {
        println "INFO: Deploying to DEV2-PHX foundation"
        staticFileDeployPCFWithManifest(pcfOrg, pcfSpace, 'api.sys.dev2.phx.pcf.syfbank.com')
        deployed = true
    }

    if (dev1east1 == "YES") {
        println "INFO: Deploying to DEV1-EAST1 foundation"
        staticFileDeployPCFWithManifest(pcfOrg, pcfSpace, 'api.sys.dev1.use1.pcf.syfbank.com')
        deployed = true
    }

    if (dev1east2 == "YES") {
        println "INFO: Deploying to DEV1-EAST2 foundation"
        staticFileDeployPCFWithManifest(pcfOrg, pcfSpace, 'api.sys.dev1.use2.pcf.syfbank.com')
        deployed = true
    }

    // If nothing is available, throw error
    if (!deployed) {
        error "ERROR: No PCF foundation is available for deployment."
    }
}
